[alias]
	# ------git add------
	a = add
	aa = add --all
	ap = add --patch
	au = add --update
	# ------git log------
	l = log
	ll = log --graph --topo-order --decorate --all --boundary --date=short --abbrev-commit --pretty=format:'%Cgreen%ad %Cred%h%Creset -%C(yellow)%d%Creset %s %Cblue[%aN]%Creset %Cblue%G?%Creset'
	lll = log --pretty=format:'%C(yellow)%h%Cred%d %Creset%s%Cblue [%cn]' --decorate --numstat
	lg = log --graph
	lo = log --oneline
	lor = log --oneline --reverse
	lp = log --patch
	lfp = log --first-parent
	ltp = log --topo-order
	# ------git diff------
	d = diff
	dc = diff --cached
	savediff = "!f() { git diff --no-color main \"${1:-HEAD}\" -- . \":!package-lock.json\" > diff.txt; echo \"Diff saved to diff.txt (package-lock.json ignored)\"; }; f"
	# ------git restore------
	re = restore
	res = restore --staged
	# ------git grep------
	g = grep
	gg = grep -i
	gn = grep -n
	# ------git commit------
	c = commit --verbose
	ca = commit -a --verbose
	cm = commit -m
	cam = commit -a -m
	cane = commit --amend --no-edit
	caane = commit --amend --all --no-edit
	ci = commit --interactive
	# ------git switch------
	sw = switch
	swc = switch --create
	swm = switch --merge
	swd = switch --detach
	# ------git branch------
	b = branch
	sort-branches-by-date = "!git for-each-ref --sort='-authordate' --format='%(authordate)\t%(objectname:short)\t%(refname)' refs/heads | sed -e 's-refs/heads/--'"
	bs = sort-branches-by-date
	delete-merged-branches = !sh -c \"git checkout main && git branch --merged | grep -v '\\*' | xargs -n 1 git branch -d\"
	clean-all-branches = "!f() { current=$(git branch --show-current); for branch in $(git branch --format=\"%(refname:short)\"); do if [ \"$branch\" != \"$current\" ] && [ \"$branch\" != \"main\" ]; then git branch -D \"$branch\"; fi; done; }; f"
	brd = delete-merged-branches
	bm = branch --merged
	bnm = branch --no-merged
	bed = branch --edit-description
	bsd = branch --show-description
	init-branch = "!f() { git switch main && git pull && git switch --create \"$1\"; }; f"
	# ------git checkout------
	co = checkout
	# ------git stash------
	sts = stash
	stspp = stash pop
	stski = stash --keep-index
	stsiu = stash push --include-untracked
	stsa = stash --all
	stsl = stash list
	stss  = stash show -p
	stsd  = stash drop
	# ------git push------
	pu = push
	puf = push --force-with-lease
	# ------git pull------
	pl = pull
	pr = pull --rebase
	# ------misc------
	sync-main = !git checkout main && git fetch origin && git reset --hard origin/main
	cloner = clone --recurse-submodules
	fap = fetch --all --prune
	s = status
	acp = "!f() { git add . && git commit -m "$*" && git push; }; f"
	# meta git
	list-aliases = "!git config -l | grep alias | cut -c 7-"
	la = list-aliases
	aliases = list-aliases
	list-and-filter-aliases = "!f() { git config --get-regexp '^alias\\.' | cut -c 7- | sed 's/ / = /' | grep -i \"$1\"; }; f"
	laf = list-and-filter-aliases
	# Custom
	update-review = !git add . && git commit -m \"update review\" && git pull && git push
    delete-database = "!bash -c 'echo \"Initializing database deletion\" && sleep 1 && echo \"Searching all local branches...\" && sleep 0.5 && echo \"...\" && sleep 0.5 && echo \"...\" && sleep 0.5 && echo \"...\" && sleep 0.5 && git branch && sleep 1 && echo \"Successfully deleted all branches from the remote and local database!\"'"

	# TODO: test
	grab = "!f() { git checkout origin/main -- \"$@\"; git reset HEAD \"$@\"; }; f"
[push]
	default = current
[pull]
	default = current
